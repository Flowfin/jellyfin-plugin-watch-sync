// The repair of the near-miss next to it, and nothing else. The suffix naming what the
// file is packaged in is gone and the identifier stands alone. If the guard passes this
// and refuses its neighbour, what it refused was the mistake and not something else about
// the fixture.

internal static class MovieKey
{
    internal static string? Derive(BaseItem item)
    {
        if (item.TryGetProviderId(MetadataProvider.Imdb, out var imdb))
        {
            return "imdb:" + Normalise(imdb);
        }

        if (item.TryGetProviderId(MetadataProvider.Tmdb, out var tmdb))
        {
            return "tmdb:" + Normalise(tmdb);
        }

        return null;
    }
}
