# The attest job with its subject widened to every file the release attaches, which is the
# change somebody makes while answering #385's third question in the other direction and
# without adding the sidecars that would have been the reason for it. It reads as generosity
# and it is a claim: the statement would then name the manifest and the inventory as things
# this run built, and a downloaded archive checked against it is no better off than before.
# The subject is decided beside which files carry a checksum, in docs/RELEASING.md, and it
# moves when that answer moves rather than on its own.

  attest:
    name: Attest build provenance
    needs: build
    permissions:
      id-token: write
      attestations: write
    steps:
      - name: Download the package
        uses: actions/download-artifact@3e5f45b2cfb9172054b4087a40e8e0b5a5461e7c # v8.0.1
        with:
          name: build-artifact

      - name: Attest build provenance
        uses: actions/attest-build-provenance@0f67c3f4856b2e3261c31976d6725780e5e4c373 # v4.1.1
        with:
          subject-path: "./*"
