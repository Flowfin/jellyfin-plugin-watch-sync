# The same slice with the one repair: the manifest is copied next to the package and the
# upload names the copy. Every path the upload is given is then in one directory, the
# download lands them flat, and the release job finds the archive where it looks for it.

  build:
    name: Build package
    steps:
      - name: Build the plugin package
        id: jprm
        uses: oddstr13/jellyfin-plugin-repository-manager@9497a0a499416cc572ed2e07a391d9f943a37b4d # v1.1.1
        with:
          dotnet-target: "net9.0"

      - name: Put the manifest beside the package
        id: assets
        env:
          ARTIFACT: ${{ steps.jprm.outputs.artifact }}
        run: |
          set -euo pipefail
          directory="$(dirname "${ARTIFACT}")"
          cp build.yaml "${directory}/build.yaml"
          echo "manifest=${directory}/build.yaml" >> "${GITHUB_OUTPUT}"

      - name: Check every release input exists
        env:
          ARTIFACT: ${{ steps.jprm.outputs.artifact }}
          MANIFEST: ${{ steps.assets.outputs.manifest }}
        run: |
          set -euo pipefail
          for file in "${ARTIFACT}" "${ARTIFACT}.meta.json" "${MANIFEST}"; do
            if [ ! -f "${file}" ]; then
              echo "::error::The packaging step did not produce ${file}."
              exit 1
            fi
          done

      - name: Hand the package to the later jobs
        uses: actions/upload-artifact@043fb46d1a93c77aae656e7c1c64a875d1fc6a0a # v7.0.1
        with:
          name: build-artifact
          retention-days: 30
          if-no-files-found: error
          path: |
            ${{ steps.jprm.outputs.artifact }}
            ${{ steps.jprm.outputs.artifact }}.meta.json
            ${{ steps.assets.outputs.manifest }}
