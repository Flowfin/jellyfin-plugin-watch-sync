// The repair of the near-miss next to it, and nothing else. The same sweep, asking an
// interface of this plugin's own for the pairings it may exchange with, and asking nothing
// at all about a key. If the guard passes this and refuses its neighbour, what it refused
// was the reach past the boundary rather than something else about the fixture.
//
// The name of this plugin's own interface deliberately begins the way the refused ones do.
// The rules name the four interfaces the pairing plugin declares rather than every name
// that starts like them, because a rule wide enough to catch a name that board has not
// written yet is wide enough to refuse the adapter #40 asks this board to write.

internal static class Sweep
{
    internal static IReadOnlyList<Guid> PairingsToExchangeWith(IPairingGateway pairings)
    {
        var active = new List<Guid>();

        foreach (var pairing in pairings.Active())
        {
            active.Add(pairing.Id);
        }

        return active;
    }
}
