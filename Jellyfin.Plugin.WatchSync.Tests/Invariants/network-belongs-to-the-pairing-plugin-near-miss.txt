// A fixture, not a source file, and its vocabulary is a fixture vocabulary.
//
// The mistake is the one the first attempt at #53's retry makes, and it is not a careless
// one. The peer's address is known, the server already offers a client factory, and a
// handler that posts the envelope itself is shorter than anything that goes through
// another plugin. It compiles, and on a peer that answers it works.
//
// What it costs is paid by the person watching something. This runs on the thread the
// server raised the event on, so a peer that accepts the connection and then says nothing
// holds that thread for as long as the socket is willing to wait, and what stops is
// playback on this server rather than a sync with a neighbour. The reachability probe
// underneath it is the same mistake wearing the clothes of a precaution: it is a second
// wait on the same peer, taken before the first one.
//
// It is also a reach past a boundary. The peer plane belongs to the pairing plugin, which
// holds the keys, the trust and the state a transfer is allowed under, so a call made from
// here is a call made outside all three.

using System.Net.Http;
using System.Net.Sockets;

internal sealed class PeerCall
{
    private readonly IHttpClientFactory _clients;

    internal PeerCall(IHttpClientFactory clients)
    {
        _clients = clients;
    }

    internal bool IsReachable(string host, int port)
    {
        using var probe = new TcpClient();

        probe.Connect(host, port);

        return probe.Connected;
    }

    internal bool Send(Uri peer, string envelope)
    {
        HttpClient client = _clients.CreateClient("watch-sync");

        using var request = new HttpRequestMessage(HttpMethod.Post, peer)
        {
            Content = new StringContent(envelope),
        };

        using HttpResponseMessage answer = client.Send(request);

        return answer.IsSuccessStatusCode;
    }
}
